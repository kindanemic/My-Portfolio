Java Code

Final Examination Ranking

This Java program collects the examination scores of 15 students. It checks that each score is between 0 and 100, calculates the average score, counts how many students passed, and sorts the scores from highest to lowest. It also displays the highest and lowest scores.

Source Code

// Ginagamit ito para makakuha ng input mula sa user
import java.util.Scanner;

// Pangalan ng buong program
public class FinalExaminationRanking {

    // Dito nagsisimula ang program
    public static void main(String[] args) {

        // Para makapag-input ng score ang user
        Scanner input = new Scanner(System.in);

        // Array na lalagyan ng scores ng 15 estudyante
        int[] scores = new int[15];

        // Lalagyan ng kabuuang score
        int total = 0;

        // Nagbibilang ng mga estudyanteng pumasa
        int passed = 0;


        // Uulit ito para makuha ulit ang scores ng 15 student
        for (int i = 0; i < 15; i++) {

            // Humihingi ng score sa bawat estudyante
            System.out.print("Enter the score: " + (i + 1) + ": ");

            // taga save ng score sa array
            scores[i] = input.nextInt();

            // Taga tingin kung ang score ay mas mababa sa 0 o mas mataas sa 100
            while (scores[i] < 0 || scores[i] > 100) {

                // Hihingi ulit ng score kapag invalid ang inilagay
                System.out.print("Invalid score. Enter between 0-100 only: ");

                // Tatanggap ng bagong score
                scores[i] = input.nextInt();
            }

            // taga score sa total
            total = total + scores[i];

            // Titingnan kung ang score ay 75 pataas
            if (scores[i] >= 75) {

                // Dadagdagan ng isa ang bilang ng mga pumasa
                passed++;
            }
        }


        // kumukuha ang average sa pamamagitan ng paghati ng total sa 15
        double average = (double) total / 15;


        // Simula ng pag-aayos ng scores mula pinakamataas hanggang pinakamababa
        for (int i = 0; i < 14; i++) {

            // taga compare ng magkatabing scores
            for (int j = 0; j < 14 - i; j++) {

                // Titingnan kung mas mababa ang score sa kaliwa
                if (scores[j] < scores[j + 1]) {

                    // Pansamantalang itatago ang score sa temp
                    int temp = scores[j];

                    // Ilalagay ang mas mataas na score sa kaliwa
                    scores[j] = scores[j + 1];

                    // taga lagay ng dating score sa kanan
                    scores[j + 1] = temp;
                }
            }
        }


        // nagpapakita ng title ng result
        System.out.println("\nFinal Examination Ranking");

        // Ipapakita ang average na may dalawang decimal places
        System.out.printf("Average Score: %.2f\n", average);

        // Ipapakita kung ilan ang pumasa
        System.out.println("Students Passed: " + passed);

        // Ipapakita ang pinakamataas na score
        System.out.println("Highest Score: " + scores[0]);

        // Ipapakita ang pinakamababang score
        System.out.println("Lowest Score: " + scores[14]);


        // Ipapakita ang pamagat ng listahan ng scores
        System.out.println("\nSorted Scores:");

        // Uulit para maipakita ang lahat ng 15 scores
        for (int i = 0; i < 15; i++) {

            // Ipapakita ang index at score ng bawat estudyante
            System.out.println("Index " + i + ": " + scores[i]);
        }

        // end ng code para di mag tuloy
        input.close();
    }
}
